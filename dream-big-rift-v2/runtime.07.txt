  function colorForParticle(i,phase,q){const base=PALETTE[colors[i]];let target;if(phase===0||phase===6)target=colors[i]%3===0?PALETTE[0]:PALETTE[1];else if(phase===2||phase===3)target=colors[i]===3?PALETTE[1]:colors[i]===4?PALETTE[2]:base;else target=colors[i]===4?PALETTE[2]:base;return mixColor(base,target,.58+q*.18)}
  function drawParticles(t){pctx.clearRect(0,0,W,H);const state=particleStateAt(t),a=state.a,b=state.b,q=state.q,cam=cameraAt(t),magpieTravel=smoother(31,41,t),magnet=pulse(15,20,42,49,t),riftX=W*.5+fbm1(t*.11)*12;pctx.save();pctx.globalCompositeOperation='lighter';for(let i=0;i<particleCount;i++){let x=lerp(a[i*3],b[i*3],q),y=lerp(a[i*3+1],b[i*3+1],q),z=lerp(a[i*3+2],b[i*3+2],q);const seed=seeds[i];x+=Math.sin(t*.18+seed*23)*(.8+Math.abs(z)*.012);y+=Math.cos(t*.16+seed*31)*(.8+Math.abs(z)*.009);if(state.phase===1||state.phase===2){x+=Math.sin(y*.011+t*.9+seed*9)*18*magnet;y+=Math.cos(x*.006-t*.55+seed*12)*11*magnet}if(state.phase===3||state.phase===4){const flight=state.phase===4?1-q:1;x+=magpieTravel*280*flight;y-=Math.sin(magpieTravel*Math.PI)*72*flight;x+=(y-H*.5)*Math.sin(magpieTravel*Math.PI)*.035*flight}if(state.phase===5){const tw=.65+.35*Math.sin(t*(1.1+seed*.6)+seed*52),r=sizes[i]*(.72+tw*.52)*(1+Math.abs(z)*.00055),col=colorForParticle(i,state.phase,q);let   function bezierPoint(a,b,c,d,t){const mt=1-t;return mt*mt*mt*a+3*mt*mt*t*b+3*mt*t*t*c+t*t*t*d}
